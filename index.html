const API_URL = "https://shuixian.blattwadsworth.workers.dev/chat"; const chat = document.getElementById("chat"); const input = document.getElementById("messageInput"); const sendButton = document.getElementById("sendButton"); const typing = document.getElementById("typing"); let welcome = document.getElementById("welcome"); let history = []; let sending = false; function scrollToBottom() { requestAnimationFrame(() => { chat.scrollTop = chat.scrollHeight; }); } function addMessage(role, text) { welcome?.remove(); // 只移除一次，重复调用无害 const row = document.createElement("div"); row.className = "message-row " + role; const bubble = document.createElement("div"); bubble.className = "bubble"; bubble.textContent = text; // 防 XSS row.appendChild(bubble); chat.appendChild(row); scrollToBottom(); return bubble; } function autoResize() { input.style.height = "auto"; input.style.height = Math.min(input.scrollHeight, 120) + "px"; } input.addEventListener("input", autoResize); input.addEventListener("keydown", (e) => { if (e.key === "Enter" && !e.shiftKey && !e.isComposing) { e.preventDefault(); send(); } }); sendButton.addEventListener("click", send); async function send() { if (sending) return; const text = input.value.trim(); if (!text) return; sending = true; sendButton.disabled = true; input.value = ""; autoResize(); addMessage("user", text); history.push({ role: "user", content: text }); history = history.slice(-20); // 限制上下文长度 typing.classList.add("show"); scrollToBottom(); try { const res = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: history }), }); if (!res.ok) throw new Error("HTTP " + res.status); const data = await res.json(); const reply = data.reply ?? data.message ?? data.content ?? ""; typing.classList.remove("show"); addMessage("assistant", reply || "（没有收到回复）"); history.push({ role: "assistant", content: reply }); history = history.slice(-20); } catch (err) { typing.classList.remove("show"); addMessage("assistant", "抱歉，连接出问题了，请稍后再试。"); console.error(err); } finally { sending = false; sendButton.disabled = false; input.focus(); } }
